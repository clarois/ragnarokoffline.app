// bounty-hunt -- the server side of the Bounty Marker.
//
// Header lines below are separated by TAB characters, not spaces.
//
// The bounty lives in permanent *character* variables (no prefix), so each is
// written to char_reg_num and comes back after a relog, an era switch or a
// server restart -- which is what makes progress persist:
//
//   bh_target     the marked monster's class id (its mob_db Id)
//   bh_item       the item id the player chose to hunt for
//   bh_goal       kills required for this bounty (scaled by the drop's rarity)
//   bh_progress   kills of bh_target since the bounty was set
//   bh_lucky      set to 1 by OnPCDropItemEvent when the hunted item drops
//                 naturally on this kill; consumed by OnNPCKillEvent to
//                 refresh the counter instead of incrementing it
//
// A bounty is one-shot: on payout all five variables are cleared, so a new
// Bounty Marker is needed to start another hunt.
//
// If the hunted item drops naturally before the counter runs out, the counter
// is *refreshed* to 0 (the bounty stays active). See OnPCDropItemEvent below
// for how the natural drop is detected -- it needs the pc_drop_item_event
// server extension, enabled via db/extension_db.yml in this same mod.
//
// The goal scales with the chosen drop's rate -- a rarer drop takes more kills,
// up to a cap -- from the mod's settings read at startup (.bh_scale/min/max).

// A place to buy Bounty Markers, so the mod is playable without GM commands.
// -1 as the price means "use the item_db Buy price" (2000z). You can also just
// @item Bounty_Marker.
prontera,152,187,4	shop	Bounty Broker#bh	4_F_KAFRA1,50071:-1

// A floating, invisible NPC (sprite -1) that owns the command and the counter.
-	script	bounty_ctrl	-1,{
	end;

OnInit:
	// How the kill goal is derived, from Settings -> Mods:
	//   goal = rate_scale / drop_rate, clamped to [min_kills, max_kills]
	// drop_rate is in units of 0.01%, so a 0.02% drop is rate 2. At the default
	// scale of 12000 that is 6000 kills; a 5% drop (rate 500) is 24. callfunc
	// keeps this working when the app is too old to know the settings.
	.bh_scale = callfunc("F_ModSetting", "bounty-hunt", "rate_scale", 12000);
	.bh_min   = callfunc("F_ModSetting", "bounty-hunt", "min_kills", 10);
	.bh_max   = callfunc("F_ModSetting", "bounty-hunt", "max_kills", 6000);
	if (.bh_scale < 1)
		.bh_scale = 1;
	if (.bh_min < 1)
		.bh_min = 1;
	if (.bh_max < .bh_min)
		.bh_max = .bh_min;
	// Bind @bounty for every player. bindatcmd's default atcommand level is 0,
	// so no groups.yml change is needed.
	bindatcmd "bounty", strnpcinfo(3) + "::OnBounty";
	end;

// @bounty            -- show the active bounty: target, item, kills done/needed.
// @bounty <mob id>   -- set a bounty. The client plugin sends this after a click;
//                       you can also type it yourself to test.
OnBounty:
	if (.@atcmd_numparameters < 1) {
		if (!bh_target || bh_goal < 1) {
			dispbottom "Bounty: none active. Use a Bounty Marker and click a monster.";
			end;
		}
		dispbottom "Bounty: " + getmonsterinfo(bh_target, MOB_NAME) + " -> " + getitemname(bh_item) + ".";
		dispbottom "Progress: " + bh_progress + " / " + bh_goal + " kills (" + (bh_goal - bh_progress) + " to go).";
		end;
	}

	.@mob = atoi(.@atcmd_parameters$[0]);
	if (.@mob <= 0) {
		dispbottom "Bounty: that is not a valid monster id.";
		end;
	}
	// getmobdrops fills $@MobDrop_item[] / $@MobDrop_rate[] and returns the drop
	// count, or 0 for an unknown monster. It does not clear the arrays between
	// calls, so copy exactly $@MobDrop_count entries and trust nothing past them.
	if (getmobdrops(.@mob) == 0) {
		dispbottom "That creature has nothing worth hunting.";
		end;
	}
	.@count = $@MobDrop_count;
	copyarray .@drop_item[0], $@MobDrop_item[0], .@count;
	copyarray .@drop_rate[0], $@MobDrop_rate[0], .@count;

	// Build the menu, skipping empty slots. The kill goal scales with each drop's
	// rate (rarer -> more kills, clamped) and is shown so the choice is informed.
	// .@opt_item[]/.@opt_goal[] map a menu row back to its item and goal.
	.@n = 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		if (.@drop_item[.@i] <= 0)
			continue;
		.@r = .@drop_rate[.@i];
		if (.@r < 1)
			.@r = 1;
		.@goal = .bh_scale / .@r;
		if (.@goal > .bh_max)
			.@goal = .bh_max;
		if (.@goal < .bh_min)
			.@goal = .bh_min;
		.@opt_item[.@n] = .@drop_item[.@i];
		.@opt_goal[.@n] = .@goal;
		.@pct$ = (.@r / 100) + "." + ((.@r % 100) < 10 ? "0" : "") + (.@r % 100) + "%";
		.@menu$[.@n] = getitemname(.@drop_item[.@i]) + " (" + .@pct$ + ", " + .@goal + " kills)";
		.@n++;
	}
	if (.@n == 0) {
		dispbottom "That creature has nothing worth hunting.";
		end;
	}

	mes "[Bounty Hunt]";
	mes "Place a bounty on ^0000FF" + getmonsterinfo(.@mob, MOB_NAME) + "^000000?";
	mes "Rarer drops take more kills.";
	next;
	// select() is 1-based; implode joins with ':' and adds no trailing entry.
	.@choice = select(implode(.@menu$, ":"));
	// Closing the window can hand back an out-of-range choice; never store a
	// bounty for an item that was not actually picked.
	if (.@choice < 1 || .@choice > .@n)
		close;

	bh_target = .@mob;
	bh_item = .@opt_item[.@choice - 1];
	bh_goal = .@opt_goal[.@choice - 1];
	bh_progress = 0;
	bh_lucky = 0;
	mes "[Bounty Hunt]";
	mes "Bounty set. Slay ^0000FF" + getmonsterinfo(.@mob, MOB_NAME) + "^000000 ^FF0000" + bh_goal + "^000000 times";
	mes "for a guaranteed ^0000FF" + getitemname(bh_item) + "^000000.";
	close;

// Fires per rolled drop from a mob, before its OnNPCKillEvent. Requires the
// pc_drop_item_event extension (enabled in this mod's db/extension_db.yml).
// killedrid is the mob's class id, killeddropid is the item id that rolled,
// killedbyme is 1 iff this pc dealt the killing blow (not a party mate, merc
// or homun). A drop of the hunted item from the current target on our own
// kill flips bh_lucky, and the imminent OnNPCKillEvent reads it below.
OnPCDropItemEvent:
	if (!bh_target || bh_goal < 1) end;
	if (killedrid != bh_target || killeddropid != bh_item) end;
	if (!killedbyme) end;
	bh_lucky = 1;
	end;

// Fires when any player kills a monster; killedrid is that monster's class id.
OnNPCKillEvent:
	if (!bh_target || bh_goal < 1 || killedrid != bh_target)
		end;
	// Lucky-drop refresh: the hunted item dropped from this kill, per the
	// OnPCDropItemEvent handler that ran a moment ago. The player already got
	// one for free, so the counter refreshes and the grind starts fresh --
	// the bounty stays active either way.
	if (bh_lucky) {
		bh_lucky = 0;
		bh_progress = 0;
		dispbottom "Bounty: lucky drop! " + getitemname(bh_item) + " landed -- counter refreshed to 0/" + bh_goal + ".";
		end;
	}
	bh_progress++;
	if (bh_progress < bh_goal) {
		dispbottom "Bounty: " + bh_progress + "/" + bh_goal + " " + getmonsterinfo(bh_target, MOB_NAME) + " (" + getitemname(bh_item) + ").";
		end;
	}
	// Goal reached -- the guaranteed drop. In OnNPCKillEvent the killer is the
	// attached player, so getitem hands it straight to them. The bounty then
	// clears completely; a new Bounty Marker is needed to start another hunt.
	getitem bh_item, 1;
	dispbottom "Bounty paid: " + getitemname(bh_item) + " secured! Use another Bounty Marker to start a new hunt.";
	bh_target = 0;
	bh_item = 0;
	bh_goal = 0;
	bh_progress = 0;
	bh_lucky = 0;
	end;
}
